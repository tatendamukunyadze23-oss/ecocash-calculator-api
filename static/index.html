function calculateFee() { let amount = parseFloat(document.getElementById('amount').value); let type = document.getElementById('type').value; // send, cashout, merchant if (!amount || amount <= 0) return; let ecocashFee = 0; let imtt = 0; if (type === 'send') { ecocashFee = amount * 0.013; // 1.3% imtt = amount >= 5 ? amount * 0.02 : 0; // 2% gov tax if $5+ } else if (type === 'cashout') { ecocashFee = amount * 0.017; // 1.7% imtt = 0; } else if (type === 'merchant') { ecocashFee = amount * 0.014; // 1.4% imtt = amount >= 5 ? amount * 0.02 : 0; } let totalFee = ecocashFee + imtt; let totalToPay = amount + totalFee; // what sender pays document.getElementById('result').innerHTML = `EcoCash: $${ecocashFee.toFixed(2)}
IMTT Tax: $${imtt.toFixed(2)}
Total Fee: $${totalFee.toFixed(2)}
You pay: $${totalToPay.toFixed(2)}`; }
